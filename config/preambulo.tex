% Dialect forms keep a searchable Unicode representation while preserving
% Penny's original phonetic notation visually. Either argument may contain
% a single word or a complete phrase.

  \BeginAccSupp{method=pdfstringdef,unicode,ActualText={#1}}%
    \textipa{#2}%
  \EndAccSupp{}%
}
